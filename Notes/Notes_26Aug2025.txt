import java.io.IOException;
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.time.Duration;

public class APITester {
   public static void main(String[] args) {
      HttpClient httpClient = HttpClient.newBuilder()
         .version(HttpClient.Version.HTTP_2)
         .connectTimeout(Duration.ofSeconds(10))
         .build(); 
         try {
            HttpRequest request = HttpRequest.newBuilder()
            .GET()
            .uri(URI.create("http://localhost:5000/food/1"))
            .build();                              
            HttpResponse<String> response = httpClient.send(request,
            HttpResponse.BodyHandlers.ofString()); 

         System.out.println("Status code: " + response.statusCode());                            
         System.out.println("Headers: " + response.headers().allValues("content-type"));
         System.out.println("Body: " + response.body());
      } catch (IOException | InterruptedException e) {
         e.printStackTrace();
      }
   }
}

package com.empower.demo.api;

import java.util.List;

import org.springframework.cloud.openfeign.FeignClient;
import org.springframework.web.bind.annotation.DeleteMapping;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.PutMapping;
import org.springframework.web.bind.annotation.RequestBody;

import com.empower.demo.Food;

@FeignClient(name = "food-api", url = "http://localhost:5000/food")
public interface FoodApi {

	@PostMapping
	public Food create(@RequestBody Food food);
	
	@GetMapping
	public List<Food> read();
	
	@GetMapping("/{id}")
	public Food read(@PathVariable Long id);
	
	@PutMapping("/{id}")
	public Food update(@PathVariable Long id, @RequestBody Food food);
	
	@DeleteMapping("/{id}")
	public Food delete(@PathVariable Long id);
}

package com.empower.demo;

import org.springframework.boot.SpringApplication;
import org.springframework.boot.autoconfigure.SpringBootApplication;
import org.springframework.cloud.gateway.route.RouteLocator;
import org.springframework.cloud.gateway.route.builder.RouteLocatorBuilder;
import org.springframework.context.annotation.Bean;

@SpringBootApplication
public class ApiGatewayApplication {

	@Bean
	public RouteLocator customRouteLocator(RouteLocatorBuilder builder) {
	    return builder.routes()
	      .route("r1", r -> r.host("**localhost:5000")
		        .and()
		        .path("/customer")
	        	        .uri("http://localhost:5001"))
	      .route("r2", r -> r.host("**localhost:5000")
	  	        .and()
	  	        .path("/product")
	  	        .uri("http://localhost:5002"))
	    .build();
	}
	
	public static void main(String[] args) {
		SpringApplication.run(ApiGatewayApplication.class, args);
	}

}





eureka navigation:-
steps


By default, the **Eureka Dashboard ([http://localhost:8761](http://localhost:8761))** only **lists registered services**, but those links are not clickable — it just shows the service `STATUS` page (e.g., `http://localhost:8081/actuator/info`).

If you want to **click a service name in Eureka and navigate directly to its API/UI**, here’s what you need to do:

---

## ✅ Steps to Make Eureka Dashboard Links Work

### 1. Enable Management Endpoints

Eureka dashboard uses the service’s `statusPageUrl` and `healthCheckUrl`.
Add these to your **microservice’s `application.properties`**:

```properties
spring.application.name=food-service
server.port=8081

# Management endpoint exposed for Eureka
management.endpoints.web.exposure.include=*
management.endpoint.health.show-details=always
```

---

### 2. Set Status and Health URLs in Eureka Client

```properties
# Register with Eureka
eureka.client.service-url.defaultZone=http://localhost:8761/eureka/

# Tell Eureka what to show in the dashboard links
eureka.instance.status-page-url=http://localhost:${server.port}/swagger-ui/index.html
eureka.instance.health-check-url=http://localhost:${server.port}/actuator/health
eureka.instance.home-page-url=http://localhost:${server.port}/
```

* `status-page-url` → What you see when clicking the service in Eureka (commonly Swagger UI).
* `home-page-url` → Default link for the service.
* `health-check-url` → Used by Eureka to check if your service is alive.

---

### 3. Example: Food Service

```properties
spring.application.name=food-service
server.port=8081

eureka.client.service-url.defaultZone=http://localhost:8761/eureka/

# Show Swagger page when clicking in Eureka
eureka.instance.status-page-url=http://localhost:${server.port}/swagger-ui/index.html
eureka.instance.home-page-url=http://localhost:${server.port}/
eureka.instance.health-check-url=http://localhost:${server.port}/actuator/health

management.endpoints.web.exposure.include=*
```

---

### 4. Run Everything

1. Start **Eureka Server (8761)**.
2. Start **food-service (8081)** and **user-service (8082)**.
3. Open **[http://localhost:8761](http://localhost:8761)**.
4. Now when you click the service link → it will take you to `http://localhost:8081/swagger-ui/index.html` (or whatever you configured).

---

✅ That way, Eureka dashboard becomes your **service navigation hub**.
👉 Pro tip: If you configure `status-page-url` for each service to point to Swagger, you’ll have a **clickable API documentation hub** right from Eureka.
