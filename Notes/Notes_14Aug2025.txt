Demo:
get an input string from the user
and extract only mobile numbers from that string.

import java.util.Scanner;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

public class App {

	public static void main(String[] args) {
		Scanner sc=new Scanner(System.in);
		String input=sc.nextLine();
		//a mobile number should start with 6/7/8/9
		//and must be only numbers
		//10 digits only
		
//		String pattern="[6-9]\\d{9}";
		Pattern p=Pattern.compile("[6-9][0-9]{9}");
		Matcher matcher = p.matcher(input);
		while(matcher.find()) {
			System.out.println(matcher.group());
		}
	}

}

package com.empower.demo;

public class Book {
	private Long id;
	private String title;
	private String author;
	private Double price;
	public Book() {
		super();
	}
	public Book(Long id, String title, String author, Double price) {
		super();
		this.id = id;
		this.title = title;
		this.author = author;
		this.price = price;
	}
	public Long getId() {
		return id;
	}
	public void setId(Long id) {
		this.id = id;
	}
	public String getTitle() {
		return title;
	}
	public void setTitle(String title) {
		this.title = title;
	}
	public String getAuthor() {
		return author;
	}
	public void setAuthor(String author) {
		this.author = author;
	}
	public Double getPrice() {
		return price;
	}
	public void setPrice(Double price) {
		this.price = price;
	}
	@Override
	public String toString() {
		return "Book [id=" + id + ", title=" + title + ", author=" + author + ", price=" + price + "]";
	}
}

interface Audio
{
	void playMusic();
}


class Sony implements Audio
{
	public void playMusic(){
	}
}

class Jbl implements Audio
{
	public void playMusic(){
	}
}




class Car{
	Sony audio;	//tightly coupled
}




class Bmw{
	Jbl audio;		//tightly coupled
}


class Car2{
	Audio audio;	//loosely coupled

	public Car2(Audio audio){
		this.audio=audio;	//constructor based injection
	}

	public void setAudio(Audio audio)
	{
		this.audio=audio;	//setter based injection
	}
}
