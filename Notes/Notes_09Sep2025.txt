		jag/******

		*/jag


2) Build triggers
	GitHub hook trigger for GITScm polling


3) Build steps

	Invoke top-level Maven targets
		goals:	clean package -DskipTests


	Execute Windows batch command

@echo off
echo Stopping old process on port 5000...

FOR /F "tokens=5" %%a IN ('netstat -ano ^| findstr :5000') DO (
    if NOT %%a==0 (
        echo Killing process %%a
        taskkill /PID %%a /F
    )
)

echo Starting new Spring Boot app in background...

for %%f in (%WORKSPACE%\hello-service\target\*.jar) do (
    echo Running %%f
    start "SpringBootApp" /b java -jar "%%f" > "%WORKSPACE%\hello-service\target\app.log" 2>&1
    goto :done
)

:done
echo Spring Boot app started. Logs: %WORKSPACE%\hello-service\target\app.log
