CREATE TABLE PRODUCTS_BACKUP
AS
SELECT * FROM PRODUCTS;


MERGE INTO PRODUCTS_BACKUP PB
USING (SELECT * FROM PRODUCTS) P
ON (PB.ID=P.ID)
WHEN MATCHED THEN
UPDATE SET PB.NAME=P.NAME, PB.PRICE=P.PRICE, PB.CATEGORY=P.CATEGORY
WHEN NOT MATCHED THEN
INSERT VALUES(P.ID, P.NAME, P.PRICE, P.CATEGORY);
